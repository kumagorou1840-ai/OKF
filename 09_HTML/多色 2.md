  <div class="card">
    <h2>プリセット切り替えテスト</h2>
    <p>ボタンを押すと背景と文字色が瞬時に切り替わり、リロード後もその状態が保持されます。</p>

  <script>
    const bgPicker = document.getElementById('bgPicker');
    const textPicker = document.getElementById('textPicker');

    // プリセットのカラー定義
    const presets = {
      light: { bg: '#ffffff', text: '#1e293b' },
      dark:  { bg: '#0f172a', text: '#f8fafc' },
      sepia: { bg: '#f4ecd8', text: '#433422' }
    };

    // 背景と文字色を適用・保存する関数
    function applyTheme(bg, text) {
      document.documentElement.style.setProperty('--bg-color', bg);
      document.documentElement.style.setProperty('--text-color', text);
      bgPicker.value = bg;
      textPicker.value = text;
      localStorage.setItem('theme_bg', bg);
      localStorage.setItem('theme_text', text);
    }

    // プリセット選択時の処理
    function setPreset(name) {
      const selected = presets[name];
      if (selected) {
        applyTheme(selected.bg, selected.text);
      }
    }

    // 手動ピッカー変更時の処理
    bgPicker.addEventListener('input', (e) => {
      document.documentElement.style.setProperty('--bg-color', e.target.value);
      localStorage.setItem('theme_bg', e.target.value);
    });

    textPicker.addEventListener('input', (e) => {
      document.documentElement.style.setProperty('--text-color', e.target.value);
      localStorage.setItem('theme_text', e.target.value);
    });

    // 初期化：localStorageから復元（未保存時はライト配色）
    const savedBg = localStorage.getItem('theme_bg') || presets.light.bg;
    const savedText = localStorage.getItem('theme_text') || presets.light.text;
    applyTheme(savedBg, savedText);
  </script>
