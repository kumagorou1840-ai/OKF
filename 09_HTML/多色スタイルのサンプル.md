<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>カラーピッカー連動サンプル</title>
  <style>
    :root {
      --theme-color: #2563eb; /* 初期値（ブルー） */
    }

    body {
      font-family: sans-serif;
      padding: 30px;
      color: #333;
    }

    /* CSS変数を利用したパーツ */
    h1 {
      color: var(--theme-color);
      border-bottom: 2px solid var(--theme-color);
      padding-bottom: 8px;
    }

    .box {
      border: 2px solid var(--theme-color);
      padding: 16px;
      margin: 16px 0;
      border-radius: 6px;
    }

    button {
      background-color: var(--theme-color);
      color: #fff;
      border: none;
      padding: 10px 18px;
      border-radius: 4px;
      cursor: pointer;
    }

    .controller {
      display: flex;
      align-items: center;
      gap: 10px;
      margin-bottom: 24px;
    }
  </style>
</head>
<body>

  <!-- カラーピッカー -->
  <div class="controller">
    <label for="colorPicker">テーマカラーを選択：</label>
    <input type="color" id="colorPicker" value="#2563eb">
    <span id="colorCode">#2563eb</span>
  </div>

  <!-- スタイル反映の確認用エリア -->
  <h1>動的テーマプレビュー</h1>
  <div class="box">
    <p>枠線や文字色に選択した色が反映されます。</p>
    <button type="button">アクションボタン</button>
  </div>

  <script>
    const picker = document.getElementById('colorPicker');
    const colorCodeText = document.getElementById('colorCode');

    // 色を動かすたびに即時反映させるため 'input' イベントを使用
    picker.addEventListener('input', (event) => {
      const selectedColor = event.target.value;

      // CSS変数を更新
      document.documentElement.style.setProperty('--theme-color', selectedColor);

      // カラーコードのテキスト表示を更新
      colorCodeText.textContent = selectedColor;
    });
  </script>

</body>
