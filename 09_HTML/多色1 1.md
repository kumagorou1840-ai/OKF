
`<input type="color">` の `input` イベントを監視し、選択されたカラーコードを `document.documentElement.style.setProperty()` でCSS変数へ即座に反映させる実装例です。


<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>カラーピッカー連動サンプル</title>
  <style>
    :root {
      --theme-color: #2563eb; /* 初期値（ブルー） */
    }

    body {
      font-family: sans-serif;
      padding: 30px;
      color: #333;
    }

    /* CSS変数を利用したパーツ */
    h1 {
      color: var(--theme-color);
      border-bottom: 2px solid var(--theme-color);
      padding-bottom: 8px;
    }

    .box {
      border: 2px solid var(--theme-color);
      padding: 16px;
      margin: 16px 0;
      border-radius: 6px;
    }

    button {
      background-color: var(--theme-color);
      color: #fff;
      border: none;
      padding: 10px 18px;
      border-radius: 4px;
      cursor: pointer;
    }

    .controller {
      display: flex;
      align-items: center;
      gap: 10px;
      margin-bottom: 24px;
    }
  </style>
</head>
<body>

  <!-- カラーピッカー -->
  <div class="controller">
    <label for="colorPicker">テーマカラーを選択：</label>
    <input type="color" id="colorPicker" value="#2563eb">
    <span id="colorCode">#2563eb</span>
  </div>

  <!-- スタイル反映の確認用エリア -->
  <h1>動的テーマプレビュー</h1>
  <div class="box">
    <p>枠線や文字色に選択した色が反映されます。</p>
    <button type="button">アクションボタン</button>
  </div>

  <script>
    const picker = document.getElementById('colorPicker');
    const colorCodeText = document.getElementById('colorCode');

    // 色を動かすたびに即時反映させるため 'input' イベントを使用
    picker.addEventListener('input', (event) => {
      const selectedColor = event.target.value;

      // CSS変数を更新
      document.documentElement.style.setProperty('--theme-color', selectedColor);

      // カラーコードのテキスト表示を更新
      colorCodeText.textContent = selectedColor;
    });
  </script>

</body>
</html>

`localStorage.setItem()` で選択した色を保存し、ページ読み込み時（初期化処理）に `localStorage.getItem()` で取得してCSS変数とカラーピッカーの値に反映させる実装コードです。

<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>localStorage連動カラー設定</title>
  <style>
    :root {

    }

      cursor: pointer;
    }
  </style>
</head>
<body>

  <div class="controller-panel">
    <div class="controller-item">
      <label for="bgPicker">背景色：</label>
      <input type="color" id="bgPicker" value="#f1f5f9">
    </div>

    <div class="controller-item">
      <label for="textPicker">文字色：</label>
      <input type="color" id="textPicker" value="#1e293b">
    </div>

    <button type="button" class="reset-btn" id="resetBtn">リセット</button>
  </div>

  <div class="card">
    <h2>設定の永続化テスト</h2>
    <p>色を変更した後にページをリロード（F5）しても、設定が保持されます。</p>
  </div>

  <script>
    const bgPicker = document.getElementById('bgPicker');
    const textPicker = document.getElementById('textPicker');
    const resetBtn = document.getElementById('resetBtn');

    const DEFAULT_BG = '#f1f5f9';
    const DEFAULT_TEXT = '#1e293b';

    // 色をCSS変数とピッカーに反映し、保存する関数
    function applyColor(type, color, save = true) {
      if (type === 'bg') {
        document.documentElement.style.setProperty('--bg-color', color);
        bgPicker.value = color;
        if (save) localStorage.setItem('theme_bg', color);
      } else if (type === 'text') {
        document.documentElement.style.setProperty('--text-color', color);
        textPicker.value = color;
        if (save) localStorage.setItem('theme_text', color);
      }
    }

    // 初期化：localStorageから読み込み（保存値がなければデフォルト値）
    const savedBg = localStorage.getItem('theme_bg') || DEFAULT_BG;
    const savedText = localStorage.getItem('theme_text') || DEFAULT_TEXT;

    applyColor('bg', savedBg, false);
    applyColor('text', savedText, false);

    // イベントリスナー：変更時に保存
    bgPicker.addEventListener('input', (e) => applyColor('bg', e.target.value));
    textPicker.addEventListener('input', (e) => applyColor('text', e.target.value));

    // リセットボタン：ストレージ削除と初期色復帰
    resetBtn.addEventListener('click', () => {
      localStorage.removeItem('theme_bg');
      localStorage.removeItem('theme_text');
      applyColor('bg', DEFAULT_BG, false);
      applyColor('text', DEFAULT_TEXT, false);
    });
  </script>

</body>
</html>
