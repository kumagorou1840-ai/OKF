    reason_str = "全必須チェック通過" if passed else " / ".join(reasons)
    return BoolAuditResult(passed, has_apology, no_careless_promise, has_action, reason_str)


def audit_scale(text: str) -> ScaleAuditResult:
    """第2層: 5段階スケール ＆ 定性ランク評価"""
    if "深くお詫び" in text or ("ご不便をおかけし" in text and "お詫び" in text):
        empathy = 5 if "深くお詫び" in text else 4
    elif "申し訳" in text or "ご不便" in text:
        empathy = 3
    else:
        empathy = 1

    if "原因を確認" in text and "協議" in text:
        risk = 5
    elif "原因を確認" in text or "協議" in text or "確認の上" in text:
        risk = 3
    elif "確認" in text:
        risk = 2
    else:
        risk = 1

    if ("時" in text or "候補" in text) and ("元栓" in text or "止水栓" in text):
        action = 5
    elif "元栓" in text or "止水栓" in text or "時" in text or "午前" in text or "午後" in text:
        action = 3
    else:
        action = 1

    avg = round((empathy + risk + action) / 3.0, 2)

    # 足切りルール: 全項目3以上 かつ 平均点によるランク分け
    if empathy >= 3 and risk >= 3 and action >= 3:
        if avg >= 4.5:
            grade = "S"
            passed = True
        elif avg >= 4.0:
            grade = "A"
            passed = True
        else:
            grade = "B"
            passed = True
    else:
        grade = "C"
        passed = False

    reason_str = f"ランク{grade}（平均 {avg} 点）: " + (
        "全項目3以上かつ高水準。" if grade in ("S", "A")
        else ("全項目3以上（標準水準・要確認推奨）。" if grade == "B" else "項目別の足切り基準（全項目3以上）を満たしていません。")
    )
    return ScaleAuditResult(passed, False, empathy, risk, action, avg, grade, reason_str)


def audit_probability(text: str, bool_res: BoolAuditResult, scale_res: ScaleAuditResult) -> ProbAuditResult:
    """第3層: 確率・不確実性（リスク・合意予測）"""
    if not bool_res.has_cause_investigation:
        financial_loss = 0.85
    elif scale_res.risk_control_level >= 4:
        financial_loss = 0.02
    elif scale_res.risk_control_level == 3:
        financial_loss = 0.12
    else:
        financial_loss = 0.35

    if scale_res.empathy_level <= 2:
        trouble_prob = 0.40
    elif scale_res.empathy_level == 3:
        trouble_prob = 0.18
    else:
        trouble_prob = 0.05

    if scale_res.action_clarity_level >= 4:
        smooth_prob = 0.92
    elif scale_res.action_clarity_level == 3:
        smooth_prob = 0.68
    else:
        smooth_prob = 0.40

    confidence = 0.94 if scale_res.grade in ("S", "A") else 0.85

    # 厳格PASS閾値（AUTO_APPROVE基準）
    passed = (trouble_prob <= 0.20) and (financial_loss <= 0.15) and (smooth_prob >= 0.70)
    if passed:
        reason_str = "リスク確率が安全閾値内（炎上<=20%, 損失<=15%, 合意>=70%）"
    elif trouble_prob <= 0.30 and financial_loss <= 0.30:
        reason_str = f"中程度リスク・合意率やや不足（炎上: {int(trouble_prob*100)}%, 損失: {int(financial_loss*100)}%, 合意: {int(smooth_prob*100)}%）"
    else:
        reason_str = f"危険リスク超過（炎上: {int(trouble_prob*100)}%, 損失: {int(financial_loss*100)}%）"

    return ProbAuditResult(passed, False, trouble_prob, financial_loss, smooth_prob, confidence, reason_str)

def run_hybrid_jev_audit(text: str, fast_fail: bool = True) -> HybridJevReport:
    # --- 第1層: 二値門番 ---
    bool_res = audit_boolean(text)

    # 早期打ち切り（Fast-Fail）発動時: 第2層・第3層をスキップして即時 AUTO_REJECT
    if fast_fail and not bool_res.passed:
        scale_res = ScaleAuditResult(
            passed=False,
            skipped=True,
            empathy_level=0,
            risk_control_level=0,
            action_clarity_level=0,
            average_score=0.0,
            grade="SKIPPED",
            reason="第1層（二値ゲート）不合格のため早期打ち切り（推論スキップ）"
        )
        prob_res = ProbAuditResult(
            passed=False,
            skipped=True,
            trouble_escalation_prob=0.50 if not bool_res.has_apology else 0.25,
            financial_loss_prob=0.85 if not bool_res.has_cause_investigation else 0.20,
            smooth_resolution_prob=0.30,
            confidence=0.98,
            reason="第1層（二値ゲート）不合格のため早期打ち切り（推論スキップ）"
        )
        verdict = "FAIL"
        action_lane = "AUTO_REJECT"
        draft_prefix = "[🛑送信不可・自動差戻し]"
        score = 0.05
        feedback = (
            f"【総合判定: {verdict} / レーン: {action_lane} / スコア: {score}】\n"
            f"・⚡ 早期打ち切り (Fast-Fail): 発動（第2・第3層をスキップ）\n"
            f"・二値判定: 不合格 ({bool_res.reason})\n"
            f"・推奨アクション: {draft_prefix} 必須項目を補完して再生成してください。"
        )
        return HybridJevReport(
            text=text,
            overall_verdict=verdict,
            action_lane=action_lane,
            draft_prefix=draft_prefix,
            fast_failed=True,
            total_score=score,
            bool_audit=asdict(bool_res),
            scale_audit=asdict(scale_res),
            prob_audit=asdict(prob_res),
            summary_feedback=feedback
        )

    # --- 第2層: 段階スケール評価 ---
    scale_res = audit_scale(text)

    # --- 第3層: 確率・確信度評価 ---
    prob_res = audit_probability(text, bool_res, scale_res)

    # --- 3レーン自動仕分け（Autonomy Slider） ---
    if (
        bool_res.passed
        and scale_res.passed
        and scale_res.grade in ("S", "A")
        and prob_res.passed
        and prob_res.confidence >= 0.90
    ):
        verdict = "PASS"
        action_lane = "AUTO_APPROVE"
        draft_prefix = "[✅Jev自動承認済]"
    elif (
        bool_res.passed
        and scale_res.grade in ("S", "A", "B")
        and prob_res.trouble_escalation_prob <= 0.30
        and prob_res.financial_loss_prob <= 0.30
    ):
        verdict = "PASS"
        action_lane = "HUMAN_REVIEW"
        draft_prefix = "[⚠️要よっちゃん確認]"
    else:
        verdict = "FAIL"
        action_lane = "AUTO_REJECT"
        draft_prefix = "[🛑送信不可・自動差戻し]"

    # 総合スコア（加重平均）
    score = (
        (scale_res.average_score / 5.0) * 0.50 +
        prob_res.smooth_resolution_prob * 0.30 +
        (1.0 if bool_res.passed else 0.0) * 0.20
    ) - (prob_res.trouble_escalation_prob * 0.20 + prob_res.financial_loss_prob * 0.20)
    score = round(max(min(score, 1.0), 0.0), 2)

    feedback = (
        f"【総合判定: {verdict} / レーン: {action_lane} ({draft_prefix}) / スコア: {score}】\n"
        f"・二値判定: {'合格' if bool_res.passed else '不合格'} ({bool_res.reason})\n"
        f"・段階判定: ランク{scale_res.grade} ({scale_res.reason})\n"
        f"・確率判定: {'合格' if prob_res.passed else '要注意/不合格'} ({prob_res.reason})"
    )

    return HybridJevReport(
        text=text,
        overall_verdict=verdict,
        action_lane=action_lane,
        draft_prefix=draft_prefix,
        fast_failed=False,
        total_score=score,
        bool_audit=asdict(bool_res),
        scale_audit=asdict(scale_res),
        prob_audit=asdict(prob_res),
        summary_feedback=feedback
    )

def call_ollama_json(prompt: str) -> dict:
    payload = {
        "model": MODEL_NAME,
        "prompt": prompt,
        "stream": False,
        "options": {"temperature": 0.1}
    }
    req = urllib.request.Request(
        "http://127.0.0.1:11434/api/generate",
        data=json.dumps(payload).encode("utf-8"),
        headers={"Content-Type": "application/json"}
    )
    with urllib.request.urlopen(req, timeout=180) as res:
        raw = json.loads(res.read().decode("utf-8")).get("response", "").strip()
    if "</think>" in raw:
        raw = raw.split("</think>")[-1].strip()
    match = re.search(r"\{.*\}", raw, re.DOTALL)
    if match:
        return json.loads(match.group(0))
    raise ValueError(f"JSON抽出失敗: {raw[:200]}")

def audit_fermi_project(project: FermiEstimationProject) -> FermiJevAudit:
    vars_list = project.variables
    total_val = project.calculate_total()

    # --- 第1層：二値（Boolean）ゲート（構造・桁ズレの確定的チェック） ---
    has_min_vars = len(vars_list) >= 3
    has_positive_vals = all(v.value > 0 for v in vars_list)
    has_rationale = all(len(v.rationale.strip()) >= 8 for v in vars_list)

    # 「率・割合」なのに 1.0 を超えている（例: 10% を 10.0 と入れて100倍桁ズレする罠）を検知
    ratio_keywords = ["率", "割合", "比率", "シェア"]
    no_percent_bug = all(
        (v.value <= 1.0)
        for v in vars_list
        if any(k in v.name or k in v.unit for k in ratio_keywords)
    )

    bool_passed = has_min_vars and has_positive_vals and has_rationale and no_percent_bug
    bool_reasons = []
    if not has_min_vars:
        bool_reasons.append("変数が2個以下で分解粒度不足（3個以上必須）")
    if not has_positive_vals:
        bool_reasons.append("0以下の不正な変数値を検知")
    if not has_rationale:
        bool_reasons.append("根拠（rationale）の説明が短すぎる変数を検知")
    if not no_percent_bug:
        bool_reasons.append("率・割合の変数が1.0を超過（%のまま掛け算し100倍桁ズレする危険）")

    bool_audit = {
        "passed": bool_passed,
        "has_min_vars_3plus": has_min_vars,
        "no_percent_100x_bug": no_percent_bug,
        "has_valid_rationale": has_rationale,
        "reason": "全必須構造チェック通過" if bool_passed else " / ".join(bool_reasons)
    }

    # --- 第2層：5段階スケール（Scale）評価 ---
    decomp_score = 5 if len(vars_list) >= 4 else (3 if len(vars_list) == 3 else 1)
    rationale_score = 5 if all(len(v.rationale) >= 14 for v in vars_list) else (3 if has_rationale else 1)
    realism_score = 5 if (no_percent_bug and 100 <= total_val <= 1e12) else (2 if no_percent_bug else 1)

    avg_scale = round((decomp_score + rationale_score + realism_score) / 3.0, 2)
    if bool_passed and decomp_score >= 3 and rationale_score >= 3 and realism_score >= 3:
        grade = "S" if avg_scale >= 4.5 else ("A" if avg_scale >= 4.0 else "B")
        scale_passed = True
    else:
        grade = "C"
        scale_passed = False

    scale_audit = {
        "passed": scale_passed,
        "decomposition_level": decomp_score,
        "rationale_clarity_level": rationale_score,
        "realism_level": realism_score,
        "average_score": avg_scale,
        "grade": grade,
        "reason": f"ランク{grade}（平均 {avg_scale} 点）"
    }

    # --- 第3層：確率（Probability）評価 ---
    order_error_prob = 0.05 if (no_percent_bug and realism_score >= 4) else 0.75
    omission_prob = 0.08 if len(vars_list) >= 4 else (0.22 if len(vars_list) == 3 else 0.60)
    confidence = 0.92 if grade in ("S", "A") else (0.80 if grade == "B" else 0.45)

    prob_passed = (order_error_prob <= 0.20) and (omission_prob <= 0.25)
    prob_audit = {
        "passed": prob_passed,
        "order_error_prob": order_error_prob,
        "omission_prob": omission_prob,
        "confidence": confidence,
        "reason": (
            f"桁ズレ確率 {int(order_error_prob*100)}% / 要素漏れ確率 {int(omission_prob*100)}% / 確信度 {int(confidence*100)}%"
        )
    }

    # --- 総合判定 ＆ 3レーン自動仕分け ---
    if bool_passed and scale_passed and grade in ("S", "A") and prob_passed:
        verdict = "PASS"
        action_lane = "AUTO_APPROVE"
        draft_prefix = "[✅Jev自動承認済]"
    elif bool_passed and scale_passed:
        verdict = "PASS"
        action_lane = "HUMAN_REVIEW"
        draft_prefix = "[⚠️要よっちゃん確認]"
    else:
        verdict = "FAIL"
        action_lane = "AUTO_REJECT"
        draft_prefix = "[🛑要再試算・差戻し]"

    total_score = round(
        max(min((avg_scale / 5.0) * 0.6 + (1.0 if bool_passed else 0.0) * 0.2 + (1.0 - order_error_prob) * 0.2, 1.0), 0.0),
        2
    )

    feedback = (
        f"判定: {verdict} ({action_lane} {draft_prefix}) / スコア: {total_score} | "
        f"二値: {bool_audit['reason']} | 段階: {scale_audit['reason']} | 確率: {prob_audit['reason']}"
    )

    return FermiJevAudit(
        verdict=verdict,
        action_lane=action_lane,
        draft_prefix=draft_prefix,
        total_score=total_score,
        bool_audit=bool_audit,
        scale_audit=scale_audit,
        prob_audit=prob_audit,
        feedback=feedback
    )

import os
import sys
from pathlib import Path

DRIVE_ROOT = Path(r"G:\マイドライブ")

def search_file(file_name):
    if not DRIVE_ROOT.exists():
        print(f"Error: {DRIVE_ROOT} が見つかりません。Googleドライブがマウントされているか確認してください。")
        return

    matches = []
    target_lower = file_name.lower()

    skip_dirs = {'.git', '.venv', 'venv', 'node_modules', '__pycache__'}
    for root, dirs, files in os.walk(DRIVE_ROOT):
        dirs[:] = [d for d in dirs if d not in skip_dirs]
        for f in files:
            if target_lower in f.lower():
                full_path = Path(root) / f
                matches.append(full_path)
                if len(matches) >= 10:
                    break
        if len(matches) >= 10:
            break

    if not matches:
        print(f"該当するファイル '{file_name}' は見つかりませんでした。")
        return

    print(f"--- 検索結果: '{file_name}' ({len(matches)}件) ---")
    for item in matches:
        stat = item.stat()
        print(f"名前: {item.name}")
        print(f"パス: {item}")
        print(f"サイズ: {stat.st_size:,} bytes")
        print("-" * 30)

if __name__ == '__main__':
    target_name = sys.argv[1] if len(sys.argv) > 1 else 'sample.csv'
    search_file(target_name)
