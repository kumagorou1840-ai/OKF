import networkx as nx
import plotly.graph_objects as go


def visualize_plotly(G, output_html="knowledge_graph.html"):
"""NetworkXのグラフオブジェクト(G)をPlotlyでHTML可視化する"""
pos = nx.spring_layout(G, dim=2, seed=42) # 2D配置（dim=3にすると3D配置も可）

# エッジ（線）の描画データ作成
edge_x, edge_y = [], []
for edge in G.edges():
x0, y0 = pos[edge[0]]
x1, y1 = pos[edge[1]]
edge_x.extend([x0, x1, None])
edge_y.extend([y0, y1, None])

edge_trace = go.Scatter(
x=edge_x,
y=edge_y,
line=dict(width=1, color="#888"),
hoverinfo="none",
mode="lines",
)

# ノード（点）の描画データ作成
node_x, node_y, node_text = [], [], []
for node in G.nodes():
x, y = pos[node]
node_x.append(x)
node_y.append(y)
node_text.append(str(node))

node_trace = go.Scatter(
x=node_x,
y=node_y,
mode="markers+text",
text=node_text,
textposition="top center",
hoverinfo="text",
marker=dict(size=12, color="#1f77b4", line=dict(width=2, color="white")),
)

# グラフの作成と保存
fig = go.Figure(
data=[edge_trace, node_trace],
layout=go.Layout(
title="OKF Knowledge Graph",
showlegend=False,
hovermode="closest",
margin=dict(b=0, l=0, r=0, t=40),
xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
yaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
),
)

# HTMLファイルとして保存
fig.write_html(output_html)
print(f"可視化ファイルを生成しました: {output_html}")
